# ERP 第一版（MVP）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: 使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 按任务逐个执行。步骤使用 checkbox（- [ ]）语法跟踪进度。

**Goal:** 交付一个可在公司局域网内稳定运行的第一版 ERP，覆盖"海外订单 → 拆算零件 → 采购 → 收货 → 领料 → 成品入库 → 出货 → 对账 → 老板看板"全流程。

**Architecture:** 前后端分离三层：React 前端 + Fastify(Node.js/TypeScript) 后端 + PostgreSQL 数据库。以销售订单为主线串联所有单据；库存变动自动记账；5 个登录角色权限在服务端强制校验。

**Tech Stack:** Node.js 20+、TypeScript 5、Fastify 4、Prisma 5、PostgreSQL 16、Zod、JWT（httpOnly cookie）、Vitest、Supertest；前端 Vite 5、React 18、Ant Design 5、React Router 6、axios。

**Spec:** docs/superpowers/specs/2026-08-20-erp-design.md（本计划依据该设计文档展开，执行者需同时阅读 spec 与本文档）

## Global Constraints

- 第一版为**单层 BOM**：成品直接由零件组成（bom 表一条记录 = 一个成品 + 一个零件 + 用量）。
- 采购拆算使用**规则 2**：采购需求 = 订单成品数量 × BOM 零件用量 − 当前可用库存，只采购缺口；缺口若为负按 0 处理；允许采购人工调整数量。
- 账期：客户**无定金、出货后 60 天付款**；系统按出货日期自动计算应收到期日。
- 成本核算：每单成本 = 该单所有零件的采购成本合计 + 可登记的其他费用；利润 = 客户累计收款 − 成本。
- 5 个登录角色，字符串枚举：boss / purchase / warehouse / sales / finance。
- 金额用 Prisma Decimal 存储（数据库 decimal(12,2)）；数量用 Int。
- 界面文案全部使用中文。
- 所有日期/时间为 UTC 存储，展示时转换为本地。
- 部署目标：Windows 常开电脑，局域网访问，开机自启，每日自动备份。

~~~ts
// backend/src/auth/token.ts
import jwt from 'jsonwebtoken'
import { config } from '../config'
export function signToken(user: { id: number; role: string }) {
  return jwt.sign({ userId: user.id, role: user.role }, config.jwtSecret, { expiresIn: '12h' })
}
export function verifyToken(token: string) {
  try {
    const p = jwt.verify(token, config.jwtSecret) as { userId: number; role: string }
    return { userId: p.userId, role: p.role }
  } catch { return null }
}
~~~

~~~ts
// backend/src/auth/guard.ts
import { FastifyReply, FastifyRequest } from 'fastify'
import { verifyToken } from './token'
export function requireRole(...roles: string[]) {
  return async (req: FastifyRequest, reply: FastifyReply) => {
    const cookie = req.headers.cookie ?? ''
    const m = cookie.match(/(?:^|;\s*)token=([^;]+)/)
    if (!m) return reply.code(401).send({ error: '未登录' })
    const payload = verifyToken(m[1])
    if (!payload) return reply.code(401).send({ error: '登录已过期' })
    if (!roles.includes(payload.role)) return reply.code(403).send({ error: '无权限' })
    ;(req as any).user = payload
  }
}
~~~

~~~ts
// backend/src/routes/auth.ts
import { FastifyInstance } from 'fastify'
import bcrypt from 'bcryptjs'
import { prisma } from '../db'
import { signToken } from '../auth/token'
import { requireRole } from '../auth/guard'

export function authRoutes(app: FastifyInstance) {
  app.post('/api/auth/login', async (req, reply) => {
    const { username, password } = req.body as any
    const user = await prisma.user.findUnique({ where: { username } })
    if (!user || !(await bcrypt.compare(password, user.passwordHash))) {
      return reply.code(401).send({ error: '用户名或密码错误' })
    }
    const token = signToken({ id: user.id, role: user.role })
    reply.setCookie('token', token, { httpOnly: true, path: '/', sameSite: 'lax', maxAge: 60 * 60 * 12 })
    return { id: user.id, username: user.username, name: user.name, role: user.role }
  })
  app.post('/api/auth/logout', async (_req, reply) => {
    reply.clearCookie('token', { path: '/' })
    return { ok: true }
  })
  app.get('/api/auth/me', { preHandler: requireRole('boss', 'purchase', 'warehouse', 'sales', 'finance') }, async (req) => {
    const { userId } = (req as any).user
    const user = await prisma.user.findUnique({ where: { id: userId } })
    return { id: user!.id, username: user!.username, name: user!.name, role: user!.role }
  })
}
~~~

~~~ts
// backend/src/domain/bom.ts
export interface BomRow { productId: number; partId: number; qty: number }

export function bomExplode(productId: number, qty: number, boms: BomRow[]) {
  const map = new Map<number, number>()
  for (const b of boms) {
    if (b.productId !== productId) continue
    map.set(b.partId, (map.get(b.partId) ?? 0) + b.qty * qty)
  }
  return [...map.entries()].map(([partId, requiredQty]) => ({ partId, requiredQty }))
}

export function computePurchaseGap(
  requirements: Array<{ partId: number; requiredQty: number }>,
  stock: Map<number, number>
) {
  return requirements.map(r => {
    const onHand = stock.get(r.partId) ?? 0
    return { partId: r.partId, gapQty: Math.max(0, r.requiredQty - onHand) }
  })
}
~~~

~~~ts
// backend/src/domain/finance.ts
export function dueDate(shippedAt: Date): Date {
  const d = new Date(shippedAt)
  d.setUTCDate(d.getUTCDate() + 60)
  return d
}

export function computeOrderCost(
  purchaseItems: Array<{ qty: number; unitPrice: number | { toNumber(): number } }>,
  otherCost: number
): number {
  const items = purchaseItems.reduce((sum, it) => {
    const price = typeof it.unitPrice === 'number' ? it.unitPrice : it.unitPrice.toNumber()
    return sum + it.qty * price
  }, 0)
  return items + otherCost
}

export function computeOrderProfit(totalReceived: number, cost: number): number {
  return totalReceived - cost
}
~~~

~~~ts
  itemId: number,
  delta: number,
  refType: string,
  refId: number
): Promise<number> {
  const stock = await tx.stock.findUnique({ where: { itemType_itemId: { itemType, itemId } } })
  const current = stock?.qtyOnHand ?? 0
  const next = current + delta
  if (next < 0) throw new Error('库存不足')
  await tx.stock.upsert({
    where: { itemType_itemId: { itemType, itemId } },
    update: { qtyOnHand: next },
    create: { itemType, itemId, qtyOnHand: next }
  })
  await tx.inventoryLedger.create({ data: { itemType, itemId, delta, balance: next, refType, refId } })
  return next
}
~~~
